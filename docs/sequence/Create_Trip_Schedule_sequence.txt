actor Admin
participant "Admin Web App" as Web
participant "AdminController" as ScheduleController
participant "AdminModel" as ScheduleRepository
database "BusRoute Collection" as RouteDB
database "FleetBus Collection" as VehicleDB
database "DriverShiftAssignment Collection" as DriverAssignDB
database "AssistantShiftAssignment Collection" as AssistantAssignDB
database "TripSchedule Collection" as ScheduleDB

note over ScheduleController,ScheduleRepository
Trip schedule creation is implemented in AdminController/AdminModel. There is no separate ScheduleService, TripController, or AssignmentController for this admin flow.
end note

Admin -> Web: 1: Submit trip schedule
activate Web
Web -> ScheduleController: 1.1: POST /api/admin/trip-schedules
activate ScheduleController
ScheduleController -> ScheduleRepository: 1.2: findRouteById(routeId) during normalizeTripSchedulePayload
activate ScheduleRepository
ScheduleRepository -> RouteDB: 1.2.1: find route
activate RouteDB
RouteDB --> ScheduleRepository: 1.2.2: route or null
deactivate RouteDB
ScheduleRepository --> ScheduleController: 1.3: route context
deactivate ScheduleRepository
ScheduleController -> ScheduleController: 1.4: validate date, time, route schedule, status
alt [route unavailable or validation failed]
    ScheduleController --> Web: 1.5.1: 400 or 409 schedule error
else [payload valid]
    opt [vehicle selected]
        ScheduleController -> ScheduleRepository: 1.6.1: findBuses()
        activate ScheduleRepository
        ScheduleRepository -> VehicleDB: 1.6.2: find FleetBus list
        activate VehicleDB
        VehicleDB --> ScheduleRepository: 1.6.3: buses
        deactivate VehicleDB
        ScheduleRepository --> ScheduleController: 1.6.4: selected vehicle availability
        deactivate ScheduleRepository
    end
    opt [driver selected]
        ScheduleController -> ScheduleRepository: 1.7.1: findEligibleDriverShiftAssignment(payload)
        activate ScheduleRepository
        ScheduleRepository -> DriverAssignDB: 1.7.2: find active driver shift assignment
        activate DriverAssignDB
        DriverAssignDB --> ScheduleRepository: 1.7.3: eligible assignment or null
        deactivate DriverAssignDB
        ScheduleRepository --> ScheduleController: 1.7.4: driver eligibility
        deactivate ScheduleRepository
    end
    opt [assistant selected]
        ScheduleController -> ScheduleRepository: 1.8.1: findEligibleAssistantShiftAssignment(payload)
        activate ScheduleRepository
        ScheduleRepository -> AssistantAssignDB: 1.8.2: find active assistant shift assignment
        activate AssistantAssignDB
        AssistantAssignDB --> ScheduleRepository: 1.8.3: eligible assignment or null
        deactivate AssistantAssignDB
        ScheduleRepository --> ScheduleController: 1.8.4: assistant eligibility
        deactivate ScheduleRepository
    end
    ScheduleController -> ScheduleRepository: 1.9: findScheduleAssignmentConflicts(payload)
    activate ScheduleRepository
    ScheduleRepository -> ScheduleDB: 1.9.1: find same-day schedule conflicts
    activate ScheduleDB
    ScheduleDB --> ScheduleRepository: 1.9.2: conflicts
    deactivate ScheduleDB
    ScheduleRepository --> ScheduleController: 1.10: conflict list
    deactivate ScheduleRepository
    alt [vehicle, driver, assistant unavailable or conflict detected]
        ScheduleController --> Web: 1.11.1: 409 assignment conflict
    else [assignment available]
        ScheduleController -> ScheduleRepository: 1.12: createTripSchedule(payload)
        activate ScheduleRepository
        ScheduleRepository -> ScheduleDB: 1.12.1: save TripSchedule
        activate ScheduleDB
        ScheduleDB --> ScheduleRepository: 1.12.2: saved schedule
        deactivate ScheduleDB
        ScheduleRepository --> ScheduleController: 1.13: created schedule
        deactivate ScheduleRepository
        ScheduleController --> Web: 1.14: 201 Trip schedule created successfully
    end
end
deactivate ScheduleController
Web --> Admin: 1.15: Refresh schedules
deactivate Web
