@startuml
title Deactivate Route - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as Web
participant "AdminController" as RouteController
participant "AdminModel" as RouteRepository
database "TripSchedule Collection" as ScheduleDB
database "BusRoute Collection" as RouteDB
database "RouteStation Collection" as StopDB

Admin -> Web: 1: Suspend route
activate Web
Web -> RouteController: 1.1: PATCH /api/admin/routes/:routeId/suspend
activate RouteController
RouteController -> RouteController: 1.2: validate route id
alt [invalid route id]
    RouteController --> Web: 1.3.1: 400 Invalid route id
else [valid route id]
    RouteController -> RouteRepository: 1.4: findRouteDeactivationBlockers(routeId)
    activate RouteRepository
    RouteRepository -> ScheduleDB: 1.4.1: find active or future planned trips
    activate ScheduleDB
    ScheduleDB --> RouteRepository: 1.4.2: blocking trips
    deactivate ScheduleDB
    RouteRepository --> RouteController: 1.5: blockers
    deactivate RouteRepository
    alt [active or future trips exist]
        RouteController --> Web: 1.6.1: 409 Route cannot be deactivated
    else [no blockers]
        RouteController -> RouteRepository: 1.7: deactivateRouteById(routeId, reason)
        activate RouteRepository
        RouteRepository -> RouteDB: 1.7.1: set status SUSPENDED
        activate RouteDB
        RouteDB --> RouteRepository: 1.7.2: route or null
        deactivate RouteDB
        alt [route found]
            RouteRepository -> StopDB: 1.7.3: remove published station assignments
            activate StopDB
            StopDB --> RouteRepository: 1.7.4: assignments updated
            deactivate StopDB
        end
        RouteRepository --> RouteController: 1.8: route or null
        deactivate RouteRepository
        alt [route not found]
            RouteController --> Web: 1.9.1: 404 Route not found
        else [route deactivated]
            RouteController --> Web: 1.9.2: 200 Route deactivated successfully
        end
    end
end
deactivate RouteController
Web --> Admin: 1.10: Show suspend result
deactivate Web
@enduml
