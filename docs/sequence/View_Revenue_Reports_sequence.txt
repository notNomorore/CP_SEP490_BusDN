@startuml
title View Revenue Reports - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as AdminWebApp
participant "RevenueController" as RevenueController
participant "RevenueService" as RevenueService
participant "TransactionRepository" as TransactionRepository
database "Transaction Collection" as TransactionCollection
participant "RouteRepository" as RouteRepository
database "Route Collection" as RouteCollection
participant "AuditLogger" as AuditLogger

note over RevenueController, RevenueService
Implemented as RevenueReportController and RevenueReportService at GET /api/admin/revenue/reports.
end note

Admin -> AdminWebApp: 1: View revenue reports
activate AdminWebApp
AdminWebApp -> RevenueController: 1.1: GET /api/admin/revenue/reports
activate RevenueController
RevenueController -> RevenueService: 1.2: getRevenueReports(filters, admin)
activate RevenueService
RevenueService -> RouteRepository: 1.3: load route labels
activate RouteRepository
RouteRepository -> RouteCollection: 1.3.1: query routes
activate RouteCollection
RouteCollection --> RouteRepository: 1.3.2: route map
deactivate RouteCollection
RouteRepository --> RevenueService: 1.3.3: route labels
deactivate RouteRepository
RevenueService -> TransactionRepository: 1.4: read PromotionUsage, travelHistory, transactions
activate TransactionRepository
TransactionRepository -> TransactionCollection: 1.4.1: query revenue event sources
activate TransactionCollection
TransactionCollection --> TransactionRepository: 1.4.2: revenue events
deactivate TransactionCollection
TransactionRepository --> RevenueService: 1.4.3: filtered report events
deactivate TransactionRepository

alt [invalid filter]
    RevenueService --> RevenueController: 1.5: filter validation error
    RevenueController --> AdminWebApp: 1.6: 400 Invalid report filter
else [no data found]
    RevenueService --> RevenueController: 1.5: zero revenue report
    RevenueController --> AdminWebApp: 1.6: 200 Empty report
else [report available]
    loop 1.5: Aggregate by date, route, payment method
        RevenueService -> RevenueService: 1.5.1: build totals and breakdowns
        activate RevenueService
        RevenueService --> RevenueService: 1.5.2: grouped report section
        deactivate RevenueService
    end
    RevenueService -> AuditLogger: 1.6: log REVENUE_REPORT_VIEWED
    activate AuditLogger
    AuditLogger --> RevenueService: 1.6.1: logged
    deactivate AuditLogger
    RevenueService --> RevenueController: 1.7: revenue report response
    RevenueController --> AdminWebApp: 1.8: 200 Revenue report
end
deactivate RevenueService
deactivate RevenueController
AdminWebApp --> Admin: 1.9: Display revenue report
deactivate AdminWebApp

@enduml
