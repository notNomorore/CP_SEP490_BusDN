@startuml
title Update Route Information - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as Web
participant "AdminController" as RouteController
participant "AdminModel" as RouteRepository
database "BusRoute Collection" as RouteDB
database "RouteStation Collection" as StopDB

Admin -> Web: 1: Save route changes
activate Web
Web -> RouteController: 1.1: PUT /api/admin/routes/:routeId
activate RouteController
RouteController -> RouteController: 1.2: validate object id and route payload
alt [invalid route id or payload]
    RouteController --> Web: 1.3.1: 400 validation error
else [payload valid]
    RouteController -> RouteRepository: 1.4: updateRouteById(routeId, payload)
    activate RouteRepository
    RouteRepository -> RouteDB: 1.4.1: findByIdAndUpdate
    activate RouteDB
    RouteDB --> RouteRepository: 1.4.2: updated route or null
    deactivate RouteDB
    alt [route found]
        RouteRepository -> StopDB: 1.5.1: syncRouteStationAssignments(route)
        activate StopDB
        StopDB --> RouteRepository: 1.5.2: assignments synced
        deactivate StopDB
    end
    RouteRepository --> RouteController: 1.6: updated route or null
    deactivate RouteRepository
    alt [route not found]
        RouteController --> Web: 1.7.1: 404 Route not found
    else [route updated]
        RouteController --> Web: 1.7.2: 200 Route updated successfully
    end
end
deactivate RouteController
Web --> Admin: 1.8: Refresh route library
deactivate Web
@enduml
