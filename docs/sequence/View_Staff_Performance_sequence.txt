title View Staff Performance - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as Web
participant "AdminController" as StaffController
participant "AdminModel" as StaffRepository
database "User Collection" as UserDB
database "TripSchedule Collection" as ScheduleDB

Admin -> Web: 1: Open staff performance page
activate Web
Web -> StaffController: 1.1: GET /api/admin/staff-performance
activate StaffController
StaffController -> StaffRepository: 1.2: getStaffPerformanceSummary()
activate StaffRepository
StaffRepository -> ScheduleDB: 1.2.1: aggregate completed trip totals
activate ScheduleDB
ScheduleDB --> StaffRepository: 1.2.2: completed trip summary
deactivate ScheduleDB
StaffRepository -> UserDB: 1.2.3: aggregate staffMetrics for DRIVER and BUS_ASSISTANT
activate UserDB
UserDB --> StaffRepository: 1.2.4: staff summary
deactivate UserDB
StaffRepository --> StaffController: 1.3: summary
deactivate StaffRepository
StaffController -> StaffRepository: 1.4: findStaffPerformanceUsers()
activate StaffRepository
StaffRepository -> UserDB: 1.4.1: find staff users sorted by performance
activate UserDB
UserDB --> StaffRepository: 1.4.2: staff members
deactivate UserDB
StaffRepository -> ScheduleDB: 1.4.3: aggregate assigned/completed trips and routes by staff
activate ScheduleDB
ScheduleDB --> StaffRepository: 1.4.4: trip stats by staff
deactivate ScheduleDB
StaffRepository --> StaffController: 1.5: enriched staff members
deactivate StaffRepository
alt [no staff members]
    StaffController --> Web: 1.6.1: 200 empty performance dataset
else [performance data available]
    StaffController --> Web: 1.6.2: 200 summary and staffMembers
end
deactivate StaffController
Web --> Admin: 1.7: Render metrics, filters, chart, and staff detail
deactivate Web
@enduml
