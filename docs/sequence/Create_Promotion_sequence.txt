@startuml
title Create Promotion - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as AdminWebApp
participant "PromotionController" as PromotionController
participant "PromotionService" as PromotionService
participant "PromotionRepository" as PromotionRepository
database "Promotion Collection" as PromotionCollection
participant "AuditLogger" as AuditLogger
database "Audit Log Collection" as AuditLogCollection

Admin -> AdminWebApp: 1: Create promotion
activate AdminWebApp
AdminWebApp -> PromotionController: 1.1: POST /api/admin/promotions
activate PromotionController
PromotionController -> PromotionService: 1.2: createPromotion(payload, admin)
activate PromotionService
PromotionService -> PromotionService: 1.3: sanitize payload and assert business rules
activate PromotionService
PromotionService --> PromotionService: 1.3.1: validated promotion payload
deactivate PromotionService
PromotionService -> PromotionRepository: 1.4: create promotion
activate PromotionRepository
PromotionRepository -> PromotionCollection: 1.4.1: insert promotion
activate PromotionCollection
PromotionCollection --> PromotionRepository: 1.4.2: created promotion or duplicate code error
deactivate PromotionCollection
PromotionRepository --> PromotionService: 1.4.3: promotion create result
deactivate PromotionRepository

alt [validation failed or duplicate promotion code]
    PromotionService --> PromotionController: 1.5: promotion creation failed
    PromotionController --> AdminWebApp: 1.6: 400 Promotion validation error
else [promotion created]
    PromotionService -> AuditLogger: 1.5: log PROMOTION_CREATED
    activate AuditLogger
    AuditLogger -> AuditLogCollection: 1.5.1: insert audit log if AuditLog model exists
    activate AuditLogCollection
    AuditLogCollection --> AuditLogger: 1.5.2: audit saved
    deactivate AuditLogCollection
    AuditLogger --> PromotionService: 1.5.3: logged
    deactivate AuditLogger
    PromotionService --> PromotionController: 1.6: created promotion
    PromotionController --> AdminWebApp: 1.7: 201 Promotion created
end
deactivate PromotionService
deactivate PromotionController
AdminWebApp --> Admin: 1.8: Show new promotion
deactivate AdminWebApp

@enduml
