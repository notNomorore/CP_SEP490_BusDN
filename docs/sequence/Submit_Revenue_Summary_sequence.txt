actor "Bus Assistant" as BusAssistant
participant "Bus Assistant App" as BusAssistantApp
participant "ShiftRevenueController" as ShiftRevenueController
participant "ShiftRevenueService" as ShiftRevenueService
participant "ShiftRevenueRepository" as ShiftRevenueRepository
database "Shift Revenue Collection" as ShiftRevenueCollection
participant "TransactionRepository" as TransactionRepository
database "Transaction Collection" as TransactionCollection
participant "AuditLogger" as AuditLogger

note over ShiftRevenueController, ShiftRevenueService
Implemented as BusAssistantController.submitRevenueSummary and BusAssistantService.submitRevenueSummary.
The saved model is RevenueSummary; no separate ShiftRevenueController file exists.
end note

BusAssistant -> BusAssistantApp: 1: Submit revenue summary
activate BusAssistantApp
BusAssistantApp -> ShiftRevenueController: 1.1: POST /api/bus-assistant/revenue-summary
activate ShiftRevenueController
ShiftRevenueController -> ShiftRevenueService: 1.2: submitRevenueSummary(payload, assistant)
activate ShiftRevenueService
ShiftRevenueService -> ShiftRevenueRepository: 1.3: verify assigned shift and existing summary
activate ShiftRevenueRepository
ShiftRevenueRepository -> ShiftRevenueCollection: 1.3.1: query RevenueSummary by shift and assistant
activate ShiftRevenueCollection
ShiftRevenueCollection --> ShiftRevenueRepository: 1.3.2: existing summary or none
deactivate ShiftRevenueCollection
ShiftRevenueRepository --> ShiftRevenueService: 1.3.3: summary validation context
deactivate ShiftRevenueRepository

alt [summary already submitted]
    ShiftRevenueService --> ShiftRevenueController: 1.4: duplicate summary error
    ShiftRevenueController --> BusAssistantApp: 1.5: 409 Summary already submitted
else [submission valid]
    ShiftRevenueService -> TransactionRepository: 1.4: calculate system revenue
    activate TransactionRepository
    TransactionRepository -> TransactionCollection: 1.4.1: aggregate completed shift transactions
    activate TransactionCollection
    TransactionCollection --> TransactionRepository: 1.4.2: system totals
    deactivate TransactionCollection
    TransactionRepository --> ShiftRevenueService: 1.4.3: revenue totals
    deactivate TransactionRepository
    ShiftRevenueService -> ShiftRevenueRepository: 1.5: save RevenueSummary and complete shift
    activate ShiftRevenueRepository
    ShiftRevenueRepository -> ShiftRevenueCollection: 1.5.1: insert summary and update shift status
    activate ShiftRevenueCollection
    ShiftRevenueCollection --> ShiftRevenueRepository: 1.5.2: saved summary
    deactivate ShiftRevenueCollection
    ShiftRevenueRepository --> ShiftRevenueService: 1.5.3: reconciliation result
    deactivate ShiftRevenueRepository
    ShiftRevenueService -> AuditLogger: 1.6: log SUBMIT_REVENUE_SUMMARY
    activate AuditLogger
    AuditLogger --> ShiftRevenueService: 1.6.1: audit logged
    deactivate AuditLogger
    ShiftRevenueService --> ShiftRevenueController: 1.7: submitted summary
    ShiftRevenueController --> BusAssistantApp: 1.8: 201 Revenue summary submitted
end
deactivate ShiftRevenueService
deactivate ShiftRevenueController
BusAssistantApp --> BusAssistant: 1.9: Show reconciliation status
deactivate BusAssistantApp
