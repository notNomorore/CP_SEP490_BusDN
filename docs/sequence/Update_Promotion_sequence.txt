@startuml
title Update Promotion - Sequence Diagram

hide footbox
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
skinparam ParticipantPadding 25
skinparam BoxPadding 10

actor Admin
participant "Admin Web App" as AdminWebApp
participant "PromotionController" as PromotionController
participant "PromotionService" as PromotionService
participant "PromotionRepository" as PromotionRepository
database "Promotion Collection" as PromotionCollection
participant "AuditLogger" as AuditLogger

Admin -> AdminWebApp: 1: Update promotion
activate AdminWebApp
AdminWebApp -> PromotionController: 1.1: PUT /api/admin/promotions/{id}
activate PromotionController
PromotionController -> PromotionService: 1.2: updatePromotion(id, payload, admin)
activate PromotionService
PromotionService -> PromotionRepository: 1.3: find promotion by id
activate PromotionRepository
PromotionRepository -> PromotionCollection: 1.3.1: query promotion
activate PromotionCollection
PromotionCollection --> PromotionRepository: 1.3.2: promotion or null
deactivate PromotionCollection
PromotionRepository --> PromotionService: 1.3.3: existing promotion
deactivate PromotionRepository

alt [promotion not found]
    PromotionService --> PromotionController: 1.4: promotion not found
    PromotionController --> AdminWebApp: 1.5: 404 Promotion not found
else [promotion found but validation failed]
    PromotionService -> PromotionService: 1.4: apply payload and assert business rules
    activate PromotionService
    PromotionService --> PromotionService: 1.4.1: validation error
    deactivate PromotionService
    PromotionService --> PromotionController: 1.5: update rejected
    PromotionController --> AdminWebApp: 1.6: 400 Promotion validation error
else [promotion updated]
    PromotionService -> PromotionRepository: 1.4: save updated promotion
    activate PromotionRepository
    PromotionRepository -> PromotionCollection: 1.4.1: update promotion fields
    activate PromotionCollection
    PromotionCollection --> PromotionRepository: 1.4.2: saved promotion
    deactivate PromotionCollection
    PromotionRepository --> PromotionService: 1.4.3: updated promotion
    deactivate PromotionRepository
    PromotionService -> AuditLogger: 1.5: log PROMOTION_UPDATED
    activate AuditLogger
    AuditLogger --> PromotionService: 1.5.1: logged
    deactivate AuditLogger
    PromotionService --> PromotionController: 1.6: updated promotion
    PromotionController --> AdminWebApp: 1.7: 200 Promotion updated
end
deactivate PromotionService
deactivate PromotionController
AdminWebApp --> Admin: 1.8: Show updated promotion
deactivate AdminWebApp

@enduml
